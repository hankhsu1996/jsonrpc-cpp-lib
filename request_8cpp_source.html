53auto Request::ValidateJson(const nlohmann::json& json_obj) -> bool {
54 if (!json_obj.is_object()) {
55 return false;
56 }
57 if (!json_obj.contains("jsonrpc") || json_obj["jsonrpc"] != "2.0") {
58 return false;
59 }
60 if (!json_obj.contains("method") || !json_obj["method"].is_string()) {
61 return false;
62 }
63
64 if (json_obj.contains("params")) {
65 const auto& params = json_obj["params"];
66 if (!params.is_array() && !params.is_object() && !params.is_null()) {
67 return false;
68 }
69 }
70
71 if (json_obj.contains("id")) {
72 const auto& id = json_obj["id"];
73 if (!id.is_string() && !id.is_number_integer()) {
74 return false;
75 }
76 }
77
78 return true;
79}
