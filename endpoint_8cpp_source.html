78auto RpcEndpoint::Shutdown() -> asio::awaitable<std::expected<void, RpcError>> {
79 if (!is_running_.exchange(false)) {
80 co_return std::expected<void, RpcError>{};
81 }

110auto RpcEndpoint::SendMethodCall(
111 std::string method, std::optional<nlohmann::json> params)
112 -> asio::awaitable<std::expected<nlohmann::json, RpcError>> {
113 if (!is_running_) {
114 co_return RpcError::UnexpectedFromCode(
115 RpcErrorCode::kClientError, "RPC endpoint is not running");
116 }
117
118 auto request_id = GetNextRequestId();
119 Request request(method, std::move(params), request_id);
120 std::string message = request.ToJson().dump();
121
122 Logger()->debug("RpcEndpoint sending message: {}", message.substr(0, 70));
123 auto pending_request = std::make_shared<PendingRequest>(endpoint_strand_);
124 asio::post(endpoint_strand_, [this, request_id, pending_request] {
125 pending_requests_[request_id] = pending_request;
126 });
127
128 auto send_result = co_await transport_->SendMessage(message);
129 if (!send_result) {
130 co_return std::unexpected(send_result.error());
131 }
132
133 auto result = co_await pending_request->GetResult();
134 if (result.contains("error")) {
135 auto err = result["error"];
136 co_return RpcError::UnexpectedFromCode(
137 RpcErrorCode::kClientError, err["message"].get<std::string>());
138 }
139
140 co_return result["result"];

199auto RpcEndpoint::ProcessMessagesLoop(asio::cancellation_slot slot)
200 -> asio::awaitable<void> {
201 auto state = co_await asio::this_coro::cancellation_state;
202
203 while (is_running_ && !state.cancelled()) {
204 auto message_result = co_await transport_->ReceiveMessage();
205 if (!message_result) {
206 Logger()->error("Receive error: {}", message_result.error().Message());
207 co_await RetryDelay(executor_);
208 continue;
209 }
210
211 auto handle_result = co_await HandleMessage(*message_result);
212 if (!handle_result) {
213 Logger()->error("Handle error: {}", handle_result.error().Message());
214 co_await RetryDelay(executor_);
215 continue;
216 }
217 }

227auto RpcEndpoint::HandleMessage(std::string message)
228 -> asio::awaitable<std::expected<void, RpcError>> {
229 Logger()->debug("RpcEndpoint handling message: {}", message.substr(0, 70));
230 const auto json_message_result =
231 nlohmann::json::parse(message, nullptr, false);
232 if (json_message_result.is_discarded()) {
233 co_return RpcError::UnexpectedFromCode(
234 RpcErrorCode::kClientError, "Failed to parse message");
235 }
236 const auto &json_message = json_message_result;
237
238 if (IsResponse(json_message)) {
239 auto response = Response::FromJson(json_message);
240 if (!response.has_value()) {
241 co_return RpcError::UnexpectedFromCode(
242 RpcErrorCode::kClientError, "Invalid response");
243 }
244 co_return co_await HandleResponse(std::move(response.value()));
245 }
246
247 if (auto response = co_await dispatcher_.DispatchRequest(message)) {
248 co_return co_await transport_->SendMessage(*response);
249 }
250
251 co_return std::expected<void, RpcError>{};
252}

254auto RpcEndpoint::HandleResponse(Response response)
255 -> asio::awaitable<std::expected<void, RpcError>> {
256 auto id_opt = response.GetId();
257 if (!id_opt || !std::holds_alternative<int64_t>(*id_opt)) {
258 co_return RpcError::UnexpectedFromCode(
259 RpcErrorCode::kClientError, "Response ID missing or not int64");
260 }
261
262 const auto id = std::get<int64_t>(*id_opt);
263
264 std::shared_ptr<PendingRequest> request;
265 bool found = false;
266
267 asio::post(endpoint_strand_, [this, id, &request, &found] {
268 auto it = pending_requests_.find(id);
269 if (it != pending_requests_.end()) {
270 request = it->second;
271 pending_requests_.erase(it);
272 found = true;
273 }
274 });
275
276 co_await asio::post(co_await asio::this_coro::executor, asio::use_awaitable);
277
278 if (!found || !request) {
279 co_return RpcError::UnexpectedFromCode(
280 RpcErrorCode::kClientError,
281 "Unknown request ID: " + std::to_string(id));
282 }
283
284 request->SetResult(response.ToJson());
285 co_return std::expected<void, RpcError>{};
286}
