61 // If already shut down, return immediately

98auto RpcEndpoint::CallMethod(
99 const std::string &method,
100 std::optional<nlohmann::json> params) -> asio::awaitable<nlohmann::json> {
101 if (!is_running_) {
102 throw std::runtime_error("RPC endpoint is not running");
103 }
104
105 // Get request ID from the executor context (thread-safe)
106 auto request_id = GetNextRequestId();
107
108 // Create the request message
109 Request request(method, std::move(params), request_id);
110 std::string message = request.Dump();
111
112 // Create a pending request
113 auto pending_request = std::make_shared<PendingRequest>(endpoint_strand_);
114
115 // Store the pending request under strand protection
116 asio::post(
117 endpoint_strand_, [this, id = request_id, req = pending_request]() {
118 pending_requests_[id] = req;
119 });
120
121 // Send the request
122 co_await transport_->SendMessage(message);
123
124 // Await the result
125 auto result = co_await pending_request->GetResult();
126
127 // Check if the result contains an error
128 if (result.contains("error")) {
129 // Extract error details
130 auto error = result["error"];
131 int code = error["code"].get<int>();
132 std::string msg = error["message"].get<std::string>();
133
134 // Report and throw
135 ReportError(static_cast<ErrorCode>(code), msg);
136 throw RpcError(static_cast<ErrorCode>(code), msg);
137 }
138
139 // Return the result
140 co_return result["result"];
141}

194auto RpcEndpoint::ProcessNextMessage() -> asio::awaitable<void> {

221auto RpcEndpoint::HandleMessage(const std::string &message)
222 -> asio::awaitable<void> {
223 try {
224 // Try to parse as a JSON object
225 auto json_message = nlohmann::json::parse(message);
226
227 // Check if it's a response
228 if (json_message.contains("id") &&
229 (json_message.contains("result") || json_message.contains("error"))) {
230 Response response(json_message);
231 co_await HandleResponse(response);
232 co_return;
233 }
234
235 // Try to handle as a request
236 auto response = co_await dispatcher_.DispatchRequest(message);
237 if (response) {
238 co_await transport_->SendMessage(*response);
239 }
240 } catch (const std::exception &e) {
241 spdlog::error("Error handling message: {}", e.what());
242 throw;
243 }
244}

246auto RpcEndpoint::HandleResponse(const Response &response)
247 -> asio::awaitable<void> {
248 // Get the request ID
249 auto id_variant = response.GetId();
250 if (!id_variant.has_value()) {
251 ReportError(ErrorCode::kInvalidRequest, "Response missing ID");
252 co_return;
253 }
254
255 // Extract the integer ID - we only support integer IDs in our implementation
256 if (!std::holds_alternative<int64_t>(*id_variant)) {
257 ReportError(ErrorCode::kInvalidRequest, "Response ID must be an integer");
258 co_return;
259 }
260
261 auto id = std::get<int64_t>(*id_variant);
262
263 // Use a local variable to capture the pending request when found
264 std::shared_ptr<PendingRequest> request;
265 bool found = false;
266
267 // Find the request in a strand-protected way
268 asio::post(endpoint_strand_, [this, id, &request, &found]() {
269 auto it = pending_requests_.find(id);
270 if (it != pending_requests_.end()) {
271 request = it->second;
272 pending_requests_.erase(it);
273 found = true;
274 }
275 });
276
277 // Wait for the post to complete
278 co_await asio::post(co_await asio::this_coro::executor, asio::use_awaitable);
279
280 if (!found || !request) {
281 ReportError(
282 ErrorCode::kInvalidRequest,
283 "Received response for unknown request ID: " + std::to_string(id));
284 co_return;
285 }
286
287 // Set the result
288 request->SetResult(response.GetJson());
289
290 co_return;
291}

293void RpcEndpoint::ScheduleRetryProcessing() {
294 spdlog::debug("Scheduling retry for message processing");
295
296 auto executor = asio::get_associated_executor(endpoint_strand_);
297 asio::steady_timer retry_timer(executor);
298 retry_timer.expires_after(std::chrono::milliseconds(100));
299
300 retry_timer.async_wait([this](const asio::error_code &ec) {
301 if (!ec && is_running_) {
302 asio::co_spawn(
303 endpoint_strand_,
304 [this]() -> asio::awaitable<void> { co_await ProcessNextMessage(); },
305 asio::detached);
306 }
307 });
308}
