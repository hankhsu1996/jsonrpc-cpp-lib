10auto ValidateResponse(const nlohmann::json& json) -> bool {
11 if (!json.contains("jsonrpc") || json["jsonrpc"] != "2.0") {
12 return false;
13 }
14
15 if (json.contains("id")) {
16 bool has_result = json.contains("result");
17 bool has_error = json.contains("error");
18
19 if (has_result == has_error) { // Must have exactly one
20 return false;
21 }
22
23 if (has_error) {
24 const auto& error = json["error"];
25 if (!error.contains("code") || !error["code"].is_number() ||
26 !error.contains("message") || !error["message"].is_string()) {
27 return false;
28 }
29 }
30 }
31
32 return true;
33}

127void RpcEndpoint::HandleMessage(const std::string& message) {
128 try {
129 auto json_message = nlohmann::json::parse(message);
130 if (!ValidateResponse(json_message)) {
131 // If not a valid response, try dispatching as request
132 auto response = dispatcher_->DispatchRequest(message);
133 if (response) {
134 WithTransport([&response](transport::Transport& transport) {
135 transport.SendMessage(*response);
136 });
137 }
138 return;
139 }
140
141 // Handle response
142 if (json_message.contains("id")) {
143 RequestId request_id;
144 const auto& id = json_message["id"];
145
146 if (id.is_number()) {
147 request_id = id.get<int64_t>();
148 } else if (id.is_string()) {
149 request_id = id.get<std::string>();
150 } else {
151 return;
152 }
153
154 std::lock_guard<std::mutex> lock(pending_requests_mutex_);
155 auto it = pending_requests_.find(request_id);
156 if (it != pending_requests_.end()) {
157 it->second.set_value(json_message);
158 pending_requests_.erase(it);
159 } else {
160 spdlog::warn("Received response for unknown request ID: {}", message);
161 }
162 }
163 } catch (const std::exception&) {
164 // Invalid message, ignore
165 }
166}
