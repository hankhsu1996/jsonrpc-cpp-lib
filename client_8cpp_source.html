48auto Client::SendMethodCall(
49 const std::string &method, std::optional<nlohmann::json> params,
50 std::chrono::milliseconds timeout) -> nlohmann::json {
51 Request request(method, std::move(params), false, [this]() {
52 return GetNextRequestId();
53 });
54 auto future = SendRequestAsync(request);
55
56 if (future.wait_for(timeout) != std::future_status::ready) {
57 // Remove the pending request since we're timing out

60 requests_map_.erase(request.GetKey());

62 throw std::runtime_error(fmt::format(
63 "Request timed out after {} ms: {}", timeout.count(), method));
64 }
65 return future.get();
66}

89auto Client::SendRequestAsync(const Request &request)
90 -> std::future<nlohmann::json> {

114void Client::HandleResponse(const std::string &response) {
115 nlohmann::json json_response;
116 try {
117 json_response = nlohmann::json::parse(response);
118 } catch (const std::exception &e) {
119 spdlog::error("Failed to parse JSON response: {}", e.what());
120 throw std::runtime_error(
121 "Failed to parse JSON response: " + std::string(e.what()));
122 }
123
124 // Check if this is a notification (no "id" field)
125 if (json_response.contains("method")) {
126 // This is a notification from server
127 try {
128 const std::string &method = json_response["method"].get<std::string>();
129 std::lock_guard<std::mutex> lock(notification_handlers_mutex_);
130 auto it = notification_handlers_.find(method);
131 if (it != notification_handlers_.end()) {
132 // Extract params if they exist
133 nlohmann::json params = json_response.contains("params")
134 ? json_response["params"]
135 : nlohmann::json::object();
136 it->second(params);
137 } else {
138 spdlog::warn(
139 "No handler registered for notification method: {}", method);
140 }
141 return; // Exit early for notifications
142 } catch (const std::exception &e) {
143 spdlog::error("Error handling notification: {}", e.what());
144 throw std::runtime_error(
145 "Error handling notification: " + std::string(e.what()));
146 }
147 }
148
149 // Handle regular responses
150 if (ValidateResponse(json_response)) {
151 int request_id = json_response["id"].get<int>();
152
153 std::lock_guard<std::mutex> lock(requests_mutex_);
154 auto req_iter = requests_map_.find(request_id);
155 if (req_iter != requests_map_.end()) {
156 req_iter->second.set_value(json_response);
157 requests_map_.erase(req_iter);
158 } else {
159 spdlog::error("Received response for unknown request ID: {}", request_id);
160 throw std::runtime_error(
161 "Received response for unknown request ID: " +
162 std::to_string(request_id));
163 }
164 } else {
165 spdlog::error("Invalid JSON-RPC response: {}", json_response.dump());
166 throw std::runtime_error(
167 "Invalid JSON-RPC response: " + json_response.dump());
168 }
169}

171auto Client::ValidateResponse(const nlohmann::json &response) -> bool {
172 if (!response.contains("jsonrpc") || response["jsonrpc"] != "2.0") {
173 return false;
174 }
175
176 if (!response.contains("id")) {
177 return false;
178 }
179
180 bool has_result = response.contains("result");
181 bool has_error = response.contains("error");
182
183 if (has_result == has_error) { // Mutually exclusive
184 return false;
185 }
186
187 if (has_error) {
188 const nlohmann::json &error = response["error"];
189 if (!error.contains("code") || !error["code"].is_number() ||
190 !error.contains("message") || !error["message"].is_string()) {
191 return false;
192 }
193 }
194
195 return true;
196}
