283template <typename ParamsType, typename ResultType>
284auto RpcEndpoint::SendMethodCall(std::string method, ParamsType params)
285 -> asio::awaitable<ResultType> {
286 try {
287 // Convert typed params to JSON
288 nlohmann::json json_params = params;
289
290 // Call the method
291 nlohmann::json result = co_await SendMethodCall(method, json_params);
292
293 // Convert result to typed result
294 ResultType typed_result = result.get<ResultType>();
295
296 co_return typed_result;
297 } catch (const nlohmann::json::exception &ex) {
298 // Handle JSON conversion errors
299 throw RpcError(
300 ErrorCode::kInvalidParams,
301 std::string("Result conversion error: ") + ex.what());
302 }
303}

305template <typename ParamsType>
306auto RpcEndpoint::SendNotification(std::string method, ParamsType params)
307 -> asio::awaitable<void> {
308 try {
309 // Convert typed params to JSON and send
310 nlohmann::json json_params = params;
311 co_await SendNotification(method, json_params);
312 } catch (const nlohmann::json::exception &ex) {
313 // Log the error but don't propagate it (notifications are
314 // fire-and-forget)
315 spdlog::error(
316 "Failed to convert parameters for notification method '{}': {}", method,
317 ex.what());
318 } catch (const std::exception &ex) {
319 // Log any other errors that might occur
320 spdlog::error(
321 "Unexpected error in notification method '{}': {}", method, ex.what());
322 }
323}

327template <typename ParamsType, typename ResultType>
328void RpcEndpoint::RegisterMethodCall(
329 std::string method,
330 std::function<asio::awaitable<ResultType>(ParamsType)> handler) {
331 // Create a handler object and store its function object
332 // NOTE: We use a shared_ptr to ensure the handler object stays alive
333 // throughout the entire lifetime of any coroutine that might use it. This
334 // prevents the use-after-free issues that can occur with lambda captures in
335 // coroutines.
336 auto typed_handler =
337 std::make_shared<TypedMethodHandler<ParamsType, ResultType>>(
338 std::move(handler));
339
340 // Register a lambda that calls the handler object
341 RegisterMethodCall(
342 method,
343 [handler = std::move(typed_handler)](
344 std::optional<nlohmann::json> params) { return (*handler)(params); });
345}
