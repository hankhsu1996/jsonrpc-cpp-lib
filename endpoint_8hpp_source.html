194template <typename ParamsType, typename ResultType>
195void RpcEndpoint::RegisterMethodCall(
196 std::string method,
197 std::function<asio::awaitable<ResultType>(ParamsType)> handler) {
198 // Create a handler object and store its function object
199 // NOTE: We use a shared_ptr to ensure the handler object stays alive
200 // throughout the entire lifetime of any coroutine that might use it. This
201 // prevents the use-after-free issues that can occur with lambda captures in
202 // coroutines.
203 auto typed_handler =
204 std::make_shared<TypedMethodHandler<ParamsType, ResultType>>(
205 std::move(handler));
206
207 // Register a lambda that calls the handler object
208 RegisterMethodCall(
209 method,
210 [handler = std::move(typed_handler)](
211 std::optional<nlohmann::json> params) { return (*handler)(params); });
212}
