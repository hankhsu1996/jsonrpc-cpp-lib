86auto Dispatcher::DispatchBatchRequest(const nlohmann::json& request_json)
87 -> asio::awaitable<std::optional<std::string>> {
88 if (request_json.empty()) {
89 co_return Response::CreateLibError(ErrorCode::kInvalidRequest).ToStr();
90 }
91
92 std::vector<asio::awaitable<std::optional<nlohmann::json>>> pending_requests;
93 pending_requests.reserve(request_json.size());
94
95 // Queue all requests in parallel
96 for (const auto& element : request_json) {
97 // Validate individual request objects in the batch
98 if (!element.is_object() || !Request::ValidateJson(element)) {
99 // For invalid requests, create an immediate error response as a coroutine
100 pending_requests.push_back(
101 []() -> asio::awaitable<std::optional<nlohmann::json>> {
102 auto error_json = Response::CreateLibError(
103 ErrorCode::kInvalidRequest, std::nullopt)
104 .ToJson();
105 co_return error_json;
106 }());
107 } else {
108 // For valid requests, dispatch them normally
109 pending_requests.push_back(DispatchSingleRequest(element));
110 }
111 }
112
113 // Wait for all requests to complete
114 std::vector<nlohmann::json> responses;
115 for (auto& pending_request : pending_requests) {
116 auto response = co_await std::move(pending_request);
117 if (response) {
118 responses.push_back(*response);
119 }
120 }
121
122 if (responses.empty()) {
123 co_return std::nullopt;
124 }
125
126 co_return nlohmann::json(responses).dump();
127}

129auto Dispatcher::ValidateRequest(const nlohmann::json& request_json)
130 -> std::optional<Response> {
131 // We've already validated basic structure in ParseAndValidateJson
132 // Here we validate method existence and other semantics
133
134 if (!request_json.contains("method")) {
135 return Response::CreateLibError(
136 ErrorCode::kInvalidRequest, "Method is required");
137 }
138
139 const auto& method = request_json["method"];
140 if (!method.is_string()) {
141 return Response::CreateLibError(
142 ErrorCode::kInvalidRequest, "Method must be a string");
143 }
144
145 // For params, if present, must be object or array
146 if (request_json.contains("params")) {
147 const auto& params = request_json["params"];
148 if (!params.is_object() && !params.is_array() && !params.is_null()) {
149 return Response::CreateLibError(
150 ErrorCode::kInvalidParams, "Params must be object or array");
151 }
152 }
153
154 // Request is valid
155 return std::nullopt;
156}
